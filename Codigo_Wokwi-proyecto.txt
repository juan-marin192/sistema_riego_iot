#include <WiFi.h>
#include <PubSubClient.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <DHT.h>
#include <ArduinoJson.h>

// ============================================================================
// CONFIGURACION DE PINES (ESP32 DevKit v1)
// ============================================================================
// Pin analogo ADC1 GPIO34 para la lectura del sensor capacitivo v1.2
#define PIN_HUMEDAD_SUELO 34  

// Pin digital GPIO15 para la interfaz de datos del sensor ambiental DHT22
#define PIN_DHT22         15  

// Modelo del sensor ambiental
#define DHTTYPE           DHT22

// Pin digital GPIO26 para el control del rele de la bomba de agua
#define PIN_RELE_BOMBA    26  

// Pin digital GPIO2 para el LED indicador de estado del sistema
#define PIN_LED_ESTADO    2   

// ============================================================================
// PARAMETROS DEL MODELO MATEMATICO EDAFICO Y CLIMATICO (OSU, FAO & UNAL)
// ============================================================================
// Capacidad de campo estimada para sustrato de maceta de hogar (% VWC)
const float CAPACIDAD_CAMPO_FC = 35.0; 

// Punto de marchitez permanente para sustrato de maceta (% VWC)
const float PUNTO_MARCHITEZ_PMP = 15.0; 

// Depreciacion Maxima Permitida (MAD) para plantas de interior (40% de TAW)
const float MAD_PLANTA_HOGAR = 0.40; 

// Calculo teorico del agua total disponible (TAW = FC - PMP)
const float TAW_SUSTRATO = CAPACIDAD_CAMPO_FC - PUNTO_MARCHITEZ_PMP;

// Umbral critico de disparo de humedad volumetrica: VWC_trigger = FC - (MAD * TAW)
const float UMBRAL_VWC_TRIGGER = CAPACIDAD_CAMPO_FC - (MAD_PLANTA_HOGAR * TAW_SUSTRATO);

// Calibracion analoga del sensor capacitivo v1.2 (3.3V ADC ESP32)
const int ADC_SUELO_SECO_AIR   = 3200; // Valor ADC registrado en suelo seco
const int ADC_SUELO_SATURADO   = 1200; // Valor ADC registrado en suelo saturado

// Temporizador de seguridad: Tiempo maximo continuo de bombeo activo (10 segundos)
const unsigned long TIEMPO_MAXIMO_RIEGO_MS = 10000; 

// Intervalos temporales no bloqueantes con millis()
const unsigned long INTERVALO_LECTURA_MS   = 1000;
const unsigned long INTERVALO_PANTALLA_MS  = 1000;
const unsigned long INTERVALO_MQTT_MS      = 5000;

// Tamaño de la ventana del filtro de promedio movil (k = 5)
const int NUM_MUESTRAS_FILTRO = 5;

// ============================================================================
// CREDENCIALES DE RED Y CONFIGURACION MQTT (EMQX GUIDE)
// ============================================================================
const char* WIFI_SSID       = "Wokwi-GUEST";
const char* WIFI_PASSWORD   = "";
const char* MQTT_BROKER     = "broker.emqx.io";
const int   MQTT_PORT       = 1883;
const char* MQTT_TOPIC_DATA = "hogar/planta01/telemetria";
const char* MQTT_TOPIC_CTRL = "hogar/planta01/control";

// ============================================================================
// INSTANCIAS DE OBJETOS Y VARIABLES GLOBALES
// ============================================================================
LiquidCrystal_I2C lcd(0x27, 16, 2);
DHT dht(PIN_DHT22, DHTTYPE);
WiFiClient espClient;
PubSubClient mqttClient(espClient);

unsigned long ultimoTiempoLectura  = 0;
unsigned long ultimoTiempoPantalla = 0;
unsigned long ultimoTiempoMQTT     = 0;
unsigned long tiempoInicioRiego    = 0;

bool bombaActiva       = false;
bool bloqueoPorTimeout = false;
bool errorSensorDescon = false;

int   valorAdcFiltrado       = 0;
float humedadSueloVWC        = 0.0;
float temperaturaAire        = 0.0;
float humedadAire            = 0.0;
float vpdKilopascales        = 0.0;

int muestrasADC[NUM_MUESTRAS_FILTRO];
int indiceMuestra = 0;

// ============================================================================
// FUNCIONES AUXILIARES DE MODELADO Y CONTROL
// ============================================================================

/**
 * @brief Aplica un filtro de promedio movil de 5 muestras para eliminar ruido analogo.
 */
int aplicarFiltroPromedio(int nuevaLectura) {
  muestrasADC[indiceMuestra] = nuevaLectura;
  indiceMuestra = (indiceMuestra + 1) % NUM_MUESTRAS_FILTRO;

  long suma = 0;
  for (int i = 0; i < NUM_MUESTRAS_FILTRO; i++) {
    suma += muestrasADC[i];
  }
  return (int)(suma / NUM_MUESTRAS_FILTRO);
}

/**
 * @brief Calcula el Deficit de Presion de Vapor (VPD) segun la ecuacion FAO Penman-Monteith.
 */
float calcularVPD(float tempC, float rh) {
  if (isnan(tempC) || isnan(rh)) return 0.0;
  float es = 0.6108 * exp((17.27 * tempC) / (tempC + 237.3)); // Presion de vapor de saturacion (kPa)
  float ea = es * (rh / 100.0);                              // Presion de vapor real (kPa)
  return (es - ea);                                          // Deficit de presion de vapor (kPa)
}

/**
 * @brief Apaga la bomba inmediatamente ante cualquier evento de seguridad.
 */
void apagarBombaPorSeguridad(const char* motivo) {
  digitalWrite(PIN_RELE_BOMBA, LOW);
  bombaActiva = false;
  Serial.print("SEGURIDAD: Bomba APAGADA -> ");
  Serial.println(motivo);
}

// ============================================================================
// CONECTIVIDAD WIFI Y MQTT (PATRON EMQX)
// ============================================================================
void conectarWiFi() {
  Serial.print("Conectando a red WiFi: ");
  Serial.println(WIFI_SSID);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nConexion WiFi establecida exitosamente.");
}

void callbackMQTT(char* topic, byte* payload, unsigned int length) {
  String mensaje = "";
  for (int i = 0; i < length; i++) {
    mensaje += (char)payload[i];
  }
  Serial.print("Comando MQTT recibido en [");
  Serial.print(topic);
  Serial.print("]: ");
  Serial.println(mensaje);

  if (mensaje == "ON" && !bloqueoPorTimeout && !errorSensorDescon) {
    digitalWrite(PIN_RELE_BOMBA, HIGH);
    bombaActiva = true;
    tiempoInicioRiego = millis();
    Serial.println("INICIO RIEGO MANUAL VIA MQTT");
  } else if (mensaje == "OFF") {
    apagarBombaPorSeguridad("Comando Remoto OFF por MQTT");
  }
}

void reconectarMQTT() {
  while (!mqttClient.connected()) {
    String clientID = "ESP32_PlantaHogar_" + String(random(0xffff), HEX);
    Serial.print("Intentando conexion al Broker MQTT...");
    if (mqttClient.connect(clientID.c_str())) {
      Serial.println("Conectado al Broker MQTT.");
      mqttClient.subscribe(MQTT_TOPIC_CTRL);
    } else {
      Serial.print("Fallo conexion MQTT, rc=");
      Serial.print(mqttClient.state());
      Serial.println(" Reintentando en 5 segundos...");
      delay(5000);
    }
  }
}

void publicarTelemetriaJSON() {
  StaticJsonDocument<256> docJSON;
  docJSON["dispositivo"]    = "ESP32_Planta_Hogar";
  docJSON["humedad_vwc"]    = humedadSueloVWC;
  docJSON["umbral_trigger"] = UMBRAL_VWC_TRIGGER;
  docJSON["temperatura"]    = temperaturaAire;
  docJSON["humedad_aire"]   = humedadAire;
  docJSON["vpd_kpa"]        = vpdKilopascales;
  docJSON["bomba_estado"]   = bombaActiva ? "ON" : "OFF";
  docJSON["alerta_timeout"] = bloqueoPorTimeout;

  // Uso de String y conversion explicita a c_str() para solucionar el error de compilacion
  String jsonBuffer;
  serializeJson(docJSON, jsonBuffer);

  mqttClient.publish(MQTT_TOPIC_DATA, jsonBuffer.c_str());

  Serial.print("Publicado en MQTT [");
  Serial.print(MQTT_TOPIC_DATA);
  Serial.print("]: ");
  Serial.println(jsonBuffer);
}

// ============================================================================
// CONFIGURACION INICIAL (SETUP)
// ============================================================================
void setup() {
  Serial.begin(115200);

  pinMode(PIN_RELE_BOMBA, OUTPUT);
  pinMode(PIN_LED_ESTADO, OUTPUT);

  // FAIL-SAFE: Asegurar bomba apagada al arrancar
  digitalWrite(PIN_RELE_BOMBA, LOW);
  digitalWrite(PIN_LED_ESTADO, HIGH);

  int lecturaInicial = analogRead(PIN_HUMEDAD_SUELO);
  for (int i = 0; i < NUM_MUESTRAS_FILTRO; i++) {
    muestrasADC[i] = lecturaInicial;
  }

  dht.begin();
  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print(" RIEGO HOGAR IoT ");
  lcd.setCursor(0, 1);
  lcd.print(" MODELO MAD/VPD  ");
  delay(2000);
  lcd.clear();

  conectarWiFi();
  mqttClient.setServer(MQTT_BROKER, MQTT_PORT);
  mqttClient.setCallback(callbackMQTT);

  Serial.println("==================================================");
  Serial.println("  SISTEMA DE RIEGO HOGAR CON MODELO MAD Y VPD     ");
  Serial.println("==================================================");
}

// ============================================================================
// BUCLE PRINCIPAL (LOOP NO BLOQUEANTE)
// ============================================================================
void loop() {
  if (!mqttClient.connected()) {
    reconectarMQTT();
  }
  mqttClient.loop();

  unsigned long tiempoActual = millis();

  // --------------------------------------------------------------------------
  // BLOQUE 1: ADQUISICION, FILTRADO Y MODELADO (Cada 1 segundo)
  // --------------------------------------------------------------------------
  if (tiempoActual - ultimoTiempoLectura >= INTERVALO_LECTURA_MS) {
    ultimoTiempoLectura = tiempoActual;

    int lecturaAdcCruda = analogRead(PIN_HUMEDAD_SUELO);
    valorAdcFiltrado    = aplicarFiltroPromedio(lecturaAdcCruda);

    if (lecturaAdcCruda < 100 || lecturaAdcCruda > 4000) {
      errorSensorDescon = true;
      apagarBombaPorSeguridad("Sensor Desconectado o Cortocircuito");
    } else {
      errorSensorDescon = false;
    }

    humedadSueloVWC = map(valorAdcFiltrado, ADC_SUELO_SECO_AIR, ADC_SUELO_SATURADO, 0, 100);
    humedadSueloVWC = constrain(humedadSueloVWC, 0.0, 100.0);

    temperaturaAire = dht.readTemperature();
    humedadAire     = dht.readHumidity();
    vpdKilopascales = calcularVPD(temperaturaAire, humedadAire);

    if (humedadSueloVWC > UMBRAL_VWC_TRIGGER && bloqueoPorTimeout) {
      bloqueoPorTimeout = false;
      Serial.println("RESET: Suelo recupero humedad. Bloqueo por timeout liberado.");
    }

    // ------------------------------------------------------------------------
    // BLOQUE 2: LOGICA DECISIONAL BASADA EN UMBRAL MAD
    // ------------------------------------------------------------------------
    if (!errorSensorDescon && !bloqueoPorTimeout) {
      if (humedadSueloVWC <= UMBRAL_VWC_TRIGGER) {
        if (!bombaActiva) {
          digitalWrite(PIN_RELE_BOMBA, HIGH);
          bombaActiva = true;
          tiempoInicioRiego = tiempoActual;
          Serial.print("INICIO RIEGO AUTOMATICO: VWC (");
          Serial.print(humedadSueloVWC);
          Serial.print("%) <= Trigger MAD (");
          Serial.print(UMBRAL_VWC_TRIGGER);
          Serial.println("%)");
        } else {
          if (tiempoActual - tiempoInicioRiego >= TIEMPO_MAXIMO_RIEGO_MS) {
            bloqueoPorTimeout = true;
            apagarBombaPorSeguridad("TIMEOUT ALCANZADO (Proteccion Inundacion)");
          }
        }
      } else {
        if (bombaActiva) {
          apagarBombaPorSeguridad("Capacidad de Campo Alcanzada");
        }
      }
    }
  }

  // --------------------------------------------------------------------------
  // BLOQUE 3: TRANSMISION DE TELEMETRIA MQTT JSON (Cada 5 segundos)
  // --------------------------------------------------------------------------
  if (tiempoActual - ultimoTiempoMQTT >= INTERVALO_MQTT_MS) {
    ultimoTiempoMQTT = tiempoActual;
    publicarTelemetriaJSON();
  }

  // --------------------------------------------------------------------------
  // BLOQUE 4: ACTUALIZACION DE PANTALLA LCD (Cada 1 segundo)
  // --------------------------------------------------------------------------
  if (tiempoActual - ultimoTiempoPantalla >= INTERVALO_PANTALLA_MS) {
    ultimoTiempoPantalla = tiempoActual;

    lcd.setCursor(0, 0);
    lcd.print("VWC:");
    lcd.print((int)humedadSueloVWC);
    lcd.print("% Trig:");
    lcd.print((int)UMBRAL_VWC_TRIGGER);
    lcd.print("%");

    lcd.setCursor(0, 1);
    if (errorSensorDescon) {
      lcd.print("ERR: SENSOR OFF ");
    } else if (bloqueoPorTimeout) {
      lcd.print("ALERTA: TIMEOUT!");
    } else if (bombaActiva) {
      unsigned long tiempoTranscurrido = (millis() - tiempoInicioRiego) / 1000;
      lcd.print("REGANDO: ");
      lcd.print(tiempoTranscurrido);
      lcd.print("s    ");
    } else {
      lcd.print("VPD:");
      lcd.print(vpdKilopascales, 2);
      lcd.print("kPa OK  ");
    }
  }
}
